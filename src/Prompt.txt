_method c_celdas.celda(RnRen, RnCol)
## Objetivo: regresa una celda de acuerdo a una posicion (ren,
##           col) proporcionada
## Parámetros de entrada:
## 	Entero - RnRen - Numero de renglon en el que se encuentra la celda
## 	Entero - RnCol - Numero de columna en la que se encuentra la celda
## Parámetros de salida:   
## 	c_celda - Objeto celda solicitado 
## Regresa: c_celda - regresa un objeto celda
## Fecha de última modificación: 28/10/2004
## Autor de la modificación : dsanchez
## Descripción de la modificación: Se documento

	# Se valida que la celda solicitada exista
	_if RnRen > _self.nNumren _or RnCol > _self.nNumCol
	_then
	   _local str
	   str << RnRen.write_string + " " + RnCol.write_string
	   condition.raise(:warning, :string, "La celda que se solicita no existe " + str)	
	_endif 
	
	>> _self.collCeldas[_self.nNumCol * (RnRen-1) + RnCol ]		
	
_endmethod

_method c_celdas.new(RnRen, RnCol)
## Objetivo: Crear una instacia de celdas
## Parámetros de entrada:
## 	Entero - RnRen : Numero de renglones que tendra celdas 
## 	Entero - RnCol : Numero de columna que tendra celdas
## Parámetros de salida:   NA
## Regresa: 
## 	c_celdas - Un objeto celdas
## Fecha de última modificación: 28/10/2004
## Autor de la modificación : dsanchez
## Descripción de la modificación: Se documento

_local LnPos # Almacena temporalmente el numero de celda a utilizar

	# Asigna el numero de renglones y columnas que tendra celdas
	_self.nNumRen << RnRen
	_self.nNumCol << RnCol

	# Crea las celdas que forman parte de celdas
	_self.collCeldas <<  hash_table.new()
	_for LnRen  _over range(1, _self.nNumRen ) 
        _loop
	   _for LnCol  _over range(1, _self.nNumCol) 
	   _loop
		   LnPos << _self.nNumCol * (LnRen - 1) + LnCol
		   _self.collCeldas[LnPos] << c_celda.new(LnRen, LnCol)
	   _endloop
	_endloop
	
	_return _clone
	
_endmethod
