_method c_resumen_optico.llena_datos_celdas()
  ## 
  ##
	_local LoColorVerde << rope.new_with(0.2,0.6,0)
	_local LoColorRojo << rope.new_with(1,0,0)	
	_local renglon << 0
	_local t_serv << 0
	_local texto << ""
	_local t_total << 0

	_self.calcula_datos_celdas()

	_for iterRenglones _over .oDatos.elements()
	_loop
		_self.asigna_texto_celda(:tbl_contenido,renglon+2,1,iterRenglones[:terminal].write_string,30,:centre_centre,0,LoColorVerde)

		_for itColumna _over range(1,8)
		_loop
			_if iterRenglones[(itColumna.write_string + "_edo_constr").as_symbol()] = "E"
			_then
				
				_self.asigna_texto_celda(:tbl_contenido,renglon+2,itColumna+1,
							 iterRenglones[(itColumna.write_string).as_symbol()].write_string,
							 30,:centre_centre,0,LoColorVerde)
			_else
				_self.asigna_texto_celda(:tbl_contenido,renglon+2,itColumna+1,
							 iterRenglones[(itColumna.write_string).as_symbol()].write_string,
							 30,:centre_centre,0,LoColorRojo)
			_endif
			t_serv +<< iterRenglones[(itColumna.write_string).as_symbol()].as_number().default(0)
		_endloop
		
		_self.asigna_texto_celda(:tbl_contenido,renglon+2,10,
					 iterRenglones[:serv].write_string,30,:centre_centre,0,LoColorRojo)
		_self.asigna_texto_celda(:tbl_contenido,renglon+2,11,
					 iterRenglones[:fib].write_string,28,:centre_centre,0,LoColorRojo)
		_self.asigna_texto_celda(:tbl_contenido,renglon+2,12,
					 iterRenglones[:tot].write_string,30,:centre_centre,0,LoColorRojo)

		t_total +<< iterRenglones[:tot].as_number().default(0)

		renglon +<< 1
	_endloop

	#texto << "TOTAL SERVICIOS OPTICOS       " + t_serv.write_string + "                           " + t_total.write_string
	texto << "Total de servicios Opticos a atender " + t_serv.write_string 
	_self.asigna_texto_celda(:tbl_total,1,1,texto,30, :centre_left,0,LoColorRojo)

_method c_resumen_optico.calcula_datos_celdas()
	## 
	##1

	
	_local LoCont << 1
	.odatos << rope.new() 
	
	_if _self.elementos_bd_gis _isnt _unset _andif
	    _self.elementos_bd_gis.size > 0
	_then
		.oCedo << _self.obtener_elementos_bd_gis()[1]
	_endif
	
	##   LoTerminales <<dto.obten_terminales_fo(.ocedo.id)   
	
	
	LoCriterio << _proc(elemento1,elemento2)
			      _if elemento1.name > elemento2.name
			      _then
				      _return _false
			      _endif
			      _return _true
		 _endproc
	
	_Local LoNumDiv << 0
	_local loContadorCables << 1
	
	_for iterDivisor _over .ocedo.obtener_divisores().as_sorted_collection(LoCriterio).elements()
	_loop   
		LoNumDiv << LoNumDiv + 1
		LoRenglon << property_list.new_with(:terminal,"FO" + iterDivisor.name,
					       :|1|,"R",:|1_edo_constr|,"P",
					       :|2|,"R",:|2_edo_constr|,"P",
					       :|3|,"R",:|3_edo_constr|,"P",
					       :|4|,"R",:|4_edo_constr|,"P",
					       :|5|,"R",:|5_edo_constr|,"P",
					       :|6|,"R",:|6_edo_constr|,"P",
					       :|7|,"R",:|7_edo_constr|,"P",
					       :|8|,"R",:|8_edo_constr|,"P",
					       :serv,0,
					       :fib,0,
					       :tot,iterDivisor.output_ports().size * 8)
		

		   # Obtiene puerto de entrada de cada divisor
		   LoPuertoEntrada << IterDivisor.input_ports()[1]		   
		   # Obtiene una lista de lineas conectadas al puerto de entrada
		   LoFibraEntrada << LoPuertoEntrada.location.connected_chains()

		   ##validar LoFibraEntrada.
		   
		   # Obtiene la fibra conectada al puerto de entrada
		   LoFibras << LoFibraEntrada.an_element()

		   _if LoFibras _isnt _unset 
		   _then


		   # Obtiene la lista de pines conectados a la fibra
		   LoPines << LoFibras.connected_pins_of_types({:mit_fiber_pin})					   
		   # Obtiene el Pin Conectado al divisor
		   LoPin << LoPines.an_element()			
		   # Obtiene todos los pines conectados abajo del Pin de entrada
		   LoPinesAbajo << .oConectividad.obten_pines_conectividad_abajo(LoPin)
		   
		   # Obtiene todas las terminales de fibra conectadas a los pines
		   # y las agrega a una lista general de terminales
		   LoTerminales  << .oConectividad.obten_terminales_de_pines(LoPinesAbajo)
		
                   _for itert _over Loterminales.fast_elements()
		   _loop
			   LoCarga << 0
			   LoCargaReal  << 0
			   
			   
			           identifica_ter << itert.user!_nombre.write_string.substitute_string("FO","")
				   
				   _if identifica_ter.size > 0 _andif itert.user!_conexion = "SENCILLA"
				   _then
					   clave << identifica_ter
					   
				   _elif identifica_ter.size > 0 _andif itert.user!_conexion = "DOBLE" 
				   _then
					   clave << identifica_ter.split_by(".").an_element()
				   _endif 
				   
				   numero_en_clave << clave.numbers_and_strings[2].write_string
				   letra_en_clave  << clave.numbers_and_strings[1].write_string
				   
				   _if itert.construction_status = "EXISTENTE"
				   _then
					   LoRenglon[(numero_en_clave.write_string + "_edo_constr" ).as_symbol()] << "E"
				   _endif

				   #write("@valido", iterdivisor.name = letra_en_clave)
				   
				   _if itert.limite_fo _isnt _unset  _andif
				       itert.limite_fo.user!_carga_corto_pla_cal _isnt _unset _andif
				       itert.limite_fo.user!_carga_corto_pla_cal > 0 _andif
				       iterdivisor.name = letra_en_clave
				   _then					   
					   # des_daniel Octubre 2022
					   # Para Obtener la carga de una terminal se divide la carga
					   # calculada del limite de terminal entre el numero de
					   # terminales contenidas en el limite
					   
					   limTer << itert.limite_fo
					   LoCarga << limTer.user!_carga_corto_pla_cal.default(0)
					   LoCarga << loCarga _div limTer.terminal_distribucions.size
					   
				   
					   _if itert.user!_conexion = "DOBLE"
					   _then
						   ## Obtener primera Terminal
						   LoPriT << itert.user!_nombre.slice(4,4)
						   LuUbi  << numero_en_clave.write_string
						   
						   _if LoPriT = LuUbi
						   _then
							   ## Es la primera de las terminales
							   LoRenglon[numero_en_clave.as_symbol()] << "8"
							   LoCargaReal << LoCargaReal + 8
						   _else
							   LoRenglon[numero_en_clave.as_symbol()] << (LoCarga - 8).write_string
							   LoCargaReal << LoCargaReal + (LoCarga - 8)
						   _endif
					   _else

						  
						   LoRenglon[numero_en_clave.as_symbol()] << LoCargaReal.write_string
						   LoCargaReal << LoCargaReal + LoCarga 
						   
					   _endif
					   
		
					   LoRenglon[:serv] +<< LoCargaReal.default(0)
					   LoRenglon[numero_en_clave.as_symbol()] << LoCargaReal.write_string	   
				   _endif


				   #show(iterdivisor.name,:indice,numero_en_clave,letra_en_clave, identifica_ter,:acumulado, LoRenglon[:serv] ,:carga, LoCarga,:real, LoCargaReal  )
				   
				   
				   
				   
				   
		   _endloop
		   
		   ##obtiene lacuenta del cable principal
		   
		   LoPinPrin << .oConectividad.obten_objeto_extremo(IterDivisor.input_ports()[1])
		   
		   _if LoPinPrin _isnt _unset 
		   _then
			   (LoCabPrin,LoAtrPrin) << LoPinPrin.get_cab_and_fiber_attribute
			   cuenta_principal << LoAtrPrin.user!_cuenta
			   LoRenglon[:fib] <<  cuenta_principal
		   _endif	   


		   .oDatos.add_last(LoRenglon)
		   
		   LoCont +<< 1	
			   
		   _endif

		   
	_endloop
	
_endmethod
