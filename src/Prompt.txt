_method c_vp_plano_proy_can.DibujaLineaEntrePozoIniYFin(PoWindow,PoPlano)
	## 
	## 
	_local LoPlanoIni
	_local LoPlanoFin
	_local LcollPlanos
	_local LcollLinea
	_if _self.pozo="Inicial"
	_then
		LoPlanoIni<<PoPlano
		
		#Buscar el viewport con el pozo final
		LcollPlanos<<LoPlanoIni.layout_page.all_elements_of(c_vp_plano_proy_can)
		_for LoPln _over LcollPlanos.fast_elements()
		_loop
			_if LoPln.pozo="Final"
			_then
				LoPlanoFin<<LoPln	
			_endif
		_endloop
		_if LoPlanoIni _isnt _unset _and LoPlanoFin _isnt _unset 
		_then
			LoColor<<colour.new_from_dec("0,0,0")
			
			LoLineStyle << line_style.new_with_properties(
					       :foreground_colour,LoColor,
					       :width, 1)
			
			LcollLinea<<sector.new()

			LoTransGisVp1<<LoPlanoIni.transform
			LoTransGisVp2<<LoPlanoFin.transform

			_if (LoOE<<_self.layout_document.user!_oe) _isnt _unset
			_then
				LoTramoCobre<<LoOE.user!_tramo_cobre
				_if (LoPozo1<<LoTramoCobre.pozo1) _isnt _unset 
				_then
					_if (LoPozo2<<LoTramoCobre.pozo2) _isnt _unset 
					_then
						LoCoordPozo1<<LoPozo1.location.coord.transformed(LoTransGisVp1)
						LoCoordPozo2<<LoPozo2.location.coord.transformed(LoTransGisVp2)
						
						LcollLinea.add(LoCoordPozo1)
						LoY<<LoPlanoIni.layout_page.height-LoCoordPozo1.y-200
						LcollLinea.add(LoCoordPozo1+coordinate(0,LoY))

						LoY<<LoPlanoIni.layout_page.height-LoCoordPozo2.y-200
						LcollLinea.add(LoCoordPozo2+coordinate(0,LoY))
						LcollLinea.add(LoCoordPozo2)
						
						LcollLinea.draw_on(PoWindow,LoLineStyle)
					_endif
				_endif
			_endif
			
		_endif
	_endif

			 _for LoSec _over LoRec.route.sectors.fast_elements()

				 _for LoCoord _over LoSec.fast_elements()
